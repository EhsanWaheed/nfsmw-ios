# --- Module G2 (run 13, 8 Oct 2026; review/run13/golden-filter-plan.md): the golden colour filter ---
# Setting golden_filter (Auto follows the Look). When the grade is on (clean_grade.h), the exposure pass writes its
# alt= target GradeInput and clean_grade writes FinishInput through GradeLut (the host's 256x1 LUT built from the PC
# visual treatment object). Off: clean_grade is skipped and alt= ignored before any device call (the frame is
# today's); GradeInput is made only while the grade is on. Host input: GradeParams = (strength, 0, 0, 0).
# vt route (default; owner 8 Oct: the 360 visual treatment's stage order and formulas, our own programs): passes tagged
# grade=vt replace clean_exposure_bloom (grade=off,pc): exposure only -> GradeInput, local luminance -> GradeLuma
# (frame/4), clean_grade_vt -> FinishInput = curve + desaturation, brightness, then + bloom (ungraded, as on the 360).
# The effect vignette and the vignette darkening stay in D (after the grade). Host inputs: GradeParams = (strength,
# saturation, 0, 0), GradeC0-C3 (the cubic). TEST NFSMW_TEST_GRADE_MODE=pc = the PC-data route (grade=pc passes).
target GradeInput argb8 frame f16
target GradeLuma  argb8 frame/4

# --- Module E.3: camera motion blur + cinematic depth of field ---
# Host inputs set by the runner every frame (contract section 9), deliberately NOT
# manifest consts: DepthInfo, nearFar, cameraVelocityVS, cameraFlags, CameraFocus,
# MotionSettings. E keeps no history target (nothing to initialise after Reset).
# This block uses 6 of the 16 global consts (16 of 16 in total); the E.3 in-car
# values are per-pass constants on the clean_motion_dof pass line below.
#
# Motion: start speed m/s, saturation speed m/s, exposure s at 100% strength,
#   maximum blur length in frame heights (absolute, not scaled by strength).
#   E.2 [O]: exposure 0.05 -> 0.025 and cap 0.04 -> 0.02 (about half of E.1 at 100%).
const EMotionParams 8.0,60.0,0.025,0.02
# Full-coverage blur length px (E.2: 1 -> 3 px, keeps short blurs from softening the
#   picture), projection P22 (60 deg vertical FOV default; P11 = P22*H/W),
#   assumed depth m when depth is unavailable, cut-scene motion-blur multiplier (0 = off).
const EMotionShape  3.0,1.7320508,16.0,0.0
# Camera-rig suppression by view depth (m): chase views full<x none>y (in-car views use
#   ERigCar.xy since E.3); chase view (in region) full<z none>w.
const ERigDepth     1.2,2.0,7.0,10.0
# Chase-view player-car ellipse in uv: centre x, centre y, radius x, radius y.
const ERigRegion    0.50,0.65,0.32,0.38
# DoF (unchanged): in-focus half-width (x DoF value), transition (x DoF value),
#   unit scale to metres, blurred-copy step per blur pass in frame heights.
const EDofParams    0.5,1.0,1.0,0.0055555556
# Clear zone, chase views: r^2 clear, r^2 full, view-point uv y, residual weight inside the clear zone (all views).
#   Run 16 (owner play test 9 Oct: blur soft all over in chase view): r 0.4/1.1 at y 0.48 -> r 0.8 clear, 2.0 full,
#   anchored on the road ahead (y 0.45, the chase view's vanishing point, above the car); an ellipse (ERigCar.w).
const EMotionEdge   0.64,4.0,0.45,0.0

# Blurred quarter-size copy of FinishInput (ping-pong; the final blur lands in EDofPing).
target EDofPing argb8 frame/4 f16
target EDofPong argb8 frame/4 f16

# --- SSAO v1 (run 13, 8 Oct 2026; review/run13/ssao-ssr-design.md section 2): ambient occlusion ---
# Before every other pass: AO at frame/2 from the depth, blurred, applied to the frame, then `recopy ao` copies the
# frame again so exposure, bloom, DoF and the rest see the AO'd picture. Host inputs (clean_ao.h): AoProj, AoDepth;
# AoParams.z = setting ssao_strength / 100. Setting ssao off (the default) -> the host skips every clean_ao* pass and
# the recopy before any device call (the frame is today's); AoA/AoB are only made while ssao is on.
# AoParams: radius m, bias, strength (host), max darkening.  AoFade: fade start m, end m, max screen radius
# (frame heights), luma guard lo.  AoBlur: blur depth tolerance, luma guard hi, TEST view (host), TEST normals (host).
const AoParams 1.0,0.03,1.0,0.35
const AoFade   60.0,100.0,0.08,0.55
const AoBlur   0.05,0.9,0.0,0.0
target AoA argb8 frame/2
target AoB argb8 frame/2
pass clean_ao -> AoA s0=depth:point blend=off write=rgba
pass clean_ao_blur_h -> AoB s0=AoA:point blend=off write=rgba
pass clean_ao_blur_v -> AoA s0=AoB:point blend=off write=rgba
pass clean_ao_apply -> frame s0=AoA:linear s1=frame:point s2=depth:point blend=off write=rgb
recopy ao

# A1-A6
# v4: one chain for A and C (x mean log luminance, y mean quartic luminance; the frame read once): 6 passes, was 12
pass clean_luma_both16 -> Luma4 s0=frame:point blend=off write=rgba
pass clean_reduce16x2 -> Luma16   s0=Luma4:point    blend=off write=rgba
pass clean_reduce16x2 -> Luma64   s0=Luma16:point   blend=off write=rgba
pass clean_reduce16x2 -> Luma128  s0=Luma64:point   blend=off write=rgba
pass clean_reduce16x2 -> LumaGrid s0=Luma128:point  blend=off write=rgba
pass clean_reduce16x2 -> LogMean  s0=LumaGrid:point blend=off write=rgba
pass clean_adapt -> AdaptHistory.cur s0=LogMean:point s1=AdaptHistory.prev:point blend=off write=r

# C1-C6: the quartic-luminance statistic is LogMean.y (merged into A1-A6 above, v4).

# B1-B3 unchanged.
pass clean_bright -> BloomBright s0=frame:linear s1=AdaptHistory.cur:point blend=off write=rgba
pass clean_blur_h -> BloomH s0=BloomBright:linear blend=off write=rgba
pass clean_blur_v -> BloomV s0=BloomH:linear blend=off write=rgba

# Final C + B composite.
# s3 supplies the quartic moment; the shader takes the fourth root internally.
pass clean_exposure_bloom -> FinishInput s0=frame:point s1=AdaptHistory.cur:point s2=BloomV:linear s3=LogMean:point blend=off write=rgba alt=GradeInput grade=off,pc
pass clean_grade -> FinishInput s0=GradeInput:point s1=GradeLut:linear blend=off write=rgba grade=pc
pass clean_exposure -> GradeInput s0=frame:point s1=AdaptHistory.cur:point s3=LogMean:point blend=off write=rgba grade=vt
pass clean_grade_down -> GradeLuma s0=GradeInput:linear blend=off write=rgba grade=vt
pass clean_grade_vt -> FinishInput s0=GradeInput:point s1=GradeLuma:linear s2=BloomV:linear blend=off write=rgba grade=vt

# Module D.1 final full-resolution finish. At FinishParams=(0,0,0,0), this is identity.
pass clean_finish -> frame s0=FinishInput:point blend=off write=rgb

# Module E.3 E1-E5: blurred quarter-size copy of the A+B+C result for DoF.
# Reads only FinishInput; writes only E targets. Two separable Gaussian iterations.
pass clean_dof_down -> EDofPing s0=FinishInput:linear blend=off write=rgba
pass clean_dof_blur_h -> EDofPong s0=EDofPing:linear blend=off write=rgba
pass clean_dof_blur_v -> EDofPing s0=EDofPong:linear blend=off write=rgba
pass clean_dof_blur_h -> EDofPong s0=EDofPing:linear blend=off write=rgba
pass clean_dof_blur_v -> EDofPing s0=EDofPong:linear blend=off write=rgba

# Module E.3 E6: motion blur + DoF composite, LAST pass, alpha-blended over D.1's frame.
# Shader alpha = E coverage (explicit alpha behaviour for this technique); RGB write mask
# keeps the frame alpha. Coverage is exactly 0 when E is inactive -> frame == D.1 output.
# Per-pass constants (E.3 [O], in-car views only, cameraFlags.y = 0):
#   EMotionEdgeCar = r^2 clear, r^2 full, view-point uv y (in-car views), ease-in amount (ALL views since run 16;
#                    1 = smoothstep(t)*t, no visible ring). Run 16: r 0.6/1.8 at y 0.48 -> r 0.8/2.0 at y 0.52
#                    (the hood view's road ahead, from above the hood up past the vanishing point), ease 1 -> 0.5.
#   ERigCar        = in-car near suppression full<x m none>y m (keeps the hood sharp),
#                    in-car strength multiplier, w = clear-zone ellipse (run 16, all views): the vertical
#                    distance counts (1 + w) times, so the zone is wider than tall (0 = the old circle).
pass clean_motion_dof -> frame s0=FinishInput:linear s1=depth:point s2=EDofPing:linear blend=alpha write=rgb EMotionEdgeCar=0.64,4.0,0.52,0.5 ERigCar=3.5,5.5,0.85,0.25
